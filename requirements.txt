import os
from dotenv import load_dotenv
from groq import Groq

load_dotenv()

# Model names change over time; check console.groq.com/docs/models
MODEL = os.getenv("GROQ_MODEL", "llama-3.3-70b-versatile")


def get_client() -> Groq:
    key = os.getenv("GROQ_API_KEY")
    if not key:  # fallback for Streamlit Cloud secrets
        try:
            import streamlit as st
            key = st.secrets["GROQ_API_KEY"]
        except Exception:
            pass
    if not key:
        raise RuntimeError("GROQ_API_KEY not found. Set it in .env or Streamlit secrets.")
    return Groq(api_key=key)


def build_prompt(field: str, level: str, duration: int, unit: str, hours_per_week: int) -> str:
    return f"""Create a detailed learning roadmap.

Field: {field}
Current level: {level}
Total time available: {duration} {unit}
Study hours per week: {hours_per_week}

Format the answer in Markdown with:
1. A short overview and the end goal
2. A phase-by-phase (or week-by-week) plan that fits the time exactly
3. For each phase: topics, hands-on mini projects, and estimated hours
4. Recommended free resources (courses, docs, books, YouTube channels)
5. A final capstone project idea
6. Tips to stay on track

Tailor the depth to a {level} learner. Skip basics they already know."""


def generate_roadmap(field, level, duration, unit, hours_per_week) -> str:
    client = get_client()
    response = client.chat.completions.create(
        model=MODEL,
        messages=[
            {"role": "system",
             "content": "You are an expert mentor who designs realistic, well-structured learning roadmaps."},
            {"role": "user",
             "content": build_prompt(field, level, duration, unit, hours_per_week)},
        ],
        temperature=0.6,
        max_tokens=3000,
    )
    return response.choices[0].message.content
